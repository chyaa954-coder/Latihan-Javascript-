    <input type="number" id="nilai">

    <button onclick="cekNilai()">Periksa Nilai</button>

    <p id="hasil"></p>

    <script>
        function cekNilai() {
            let nilai = Number(document.getElementById("nilai").value);

            if (nilai >= 75) {
                document.getElementById("hasil").innerHTML = "Lulus";
            } else {
                document.getElementById("hasil").innerHTML = "Remedial";
            }
        }
    </script>
